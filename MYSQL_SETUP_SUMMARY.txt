================================================================================
💻 BASIC USAGE
================================================================================

from mysql_connection import MySQLConnection
import os
from dotenv import load_dotenv

load_dotenv()

# Connect
db = MySQLConnection(
    host=os.getenv('MYSQL_HOST'),
    user=os.getenv('MYSQL_USER'),
    password=os.getenv('MYSQL_PASSWORD'),
    database=os.getenv('MYSQL_DATABASE')
)
db.connect()

# Query as DataFrame
df = db.query_to_dataframe("SELECT * FROM users LIMIT 10")
print(df)

# Insert data
db.execute_update("INSERT INTO users (name, email) VALUES ('John', 'john@example.com')")

# Disconnect
db.disconnect()

Insert Data:

Update Data:
  db.execute_update("UPDATE users SET email = 'new@example.com' WHERE id = 1")

Delete Data:
  db.execute_update("DELETE FROM users WHERE created_at < DATE_SUB(NOW(), INTERVAL 1 YEAR)")

Get Tables:
  tables = db.get_tables()
  print(tables)

================================================================================
🐛 TROUBLESHOOTING
================================================================================

Connection Refused:
  - Check MySQL is running: mysql -u root -p
  - Verify host/port in .env
  - Check firewall rules

Access Denied:
  - Verify credentials in .env
  - Check user permissions in MySQL

Module Not Found:
  - Install: pip install mysql-connector-python
  - Check virtual environment is activated

Timeout:
  - Increase timeout in config
  - Check network connectivity
  - Verify MySQL max_connections setting
